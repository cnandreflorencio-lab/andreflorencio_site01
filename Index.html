    function renderAgenda() {

      const id = state.agendaCalendarId || "";

      const key = state.agendaApiKey || "";


      els.agendaOpen.href = id

        ? `https://calendar.google.com/calendar/u/0?cid=${encodeURIComponent(id)}`

        : "#";


      if (id && key) fetchEvents(id, key);

      else showCalendarFallback();

    }


    function showCalendarFallback() {

      els.eventsGrid.hidden = true;

      els.eventsGrid.innerHTML = "";

      els.calendarFrame.hidden = false;


      const id = state.agendaCalendarId || "";

      const url = id

        ? `https://calendar.google.com/calendar/embed?src=${encodeURIComponent(id)}&ctz=America%2FSao_Paulo`

        : (state.agendaUrl || "");


      if (!url || url.includes("SEU_CALENDARIO")) {

        els.agendaFrame.removeAttribute("src");

        els.agendaFrame.srcdoc = `<div style="font-family:Manrope,sans-serif;padding:3rem;text-align:center;color:#6d6455;">

          Configure o Google Agenda no painel admin (botão “Agenda”).<br>

          Com a chave da API, os eventos viram cards automáticos.

        </div>`;

      } else {

        els.agendaFrame.removeAttribute("srcdoc");

        els.agendaFrame.src = url;

      }

    }


    async function fetchEvents(id, key) {

      try {

        const timeMin = new Date(); timeMin.setHours(0, 0, 0, 0);

        const timeMax = new Date(); timeMax.setFullYear(timeMax.getFullYear() + 1);


        const api = `https://www.googleapis.com/calendar/v3/calendars/${encodeURIComponent(id)}/events` +

          `?key=${encodeURIComponent(key)}` +

          `&timeMin=${timeMin.toISOString()}` +

          `&timeMax=${timeMax.toISOString()}` +

          `&singleEvents=true&orderBy=startTime&maxResults=12`;


        const res = await fetch(api);

        if (!res.ok) throw new Error("API respondeu " + res.status);

        const data = await res.json();

        const items = data.items || [];


        if (!items.length) {

          els.calendarFrame.hidden = true;

          els.eventsGrid.hidden = false;

          els.eventsGrid.innerHTML = `<div class="empty">Nenhum evento futuro na agenda.</div>`;

          return;

        }

        renderEventCards(items);

      } catch (err) {

        console.warn("Não foi possível ler os eventos. Usando calendário incorporado.", err);

        showCalendarFallback();

      }

    }


    function renderEventCards(items) {

      const fmtDay = new Intl.DateTimeFormat("pt-BR", { day: "2-digit" });

      const fmtMonth = new Intl.DateTimeFormat("pt-BR", { month: "short" });

      const fmtTime = new Intl.DateTimeFormat("pt-BR", { hour: "2-digit", minute: "2-digit" });


      els.calendarFrame.hidden = true;

      els.eventsGrid.hidden = false;


      els.eventsGrid.innerHTML = items.map((ev) => {

        const start = ev.start || {};

        const d = start.dateTime ? new Date(start.dateTime) : new Date((start.date || "") + "T12:00:00");

        const hasTime = !!start.dateTime;


        const loc = (ev.location || "").trim();

        const city = loc ? loc.split("-").pop().trim() : "Local a confirmar";


        return `

          <article class="event-card">

            <div class="event-date">

              <span class="day">${fmtDay.format(d)}</span>

              <span class="month">${fmtMonth.format(d).replace(".", "").toUpperCase()}</span>

            </div>

            <div class="event-info">

              <strong>${escapeHtml(ev.summary || "Evento")}</strong>

              <span>📍 ${escapeHtml(city)}</span>

              ${hasTime ? `<span>🕗 ${fmtTime.format(d)}</span>` : `<span>🗓 Dia todo</span>`}

            </div>

          </article>`;

      }).join("");

    }


    function editAgenda() {

      const id = prompt(

        "ID do Google Agenda (e-mail do calendário):\nEx.: andre.florencio@gmail.com",

        state.agendaCalendarId || ""

      );

      if (id === null) return;


      const key = prompt(

        "Chave da API do Google (para mostrar os cards de eventos):\n\n" +

        "Deixe em branco para usar somente o calendário incorporado.",

        state.agendaApiKey || ""

      );

      if (key === null) return;


      state.agendaCalendarId = id.trim();

      state.agendaApiKey = key.trim();

      save();

      renderAgenda();


      if (state.agendaCalendarId && !state.agendaApiKey) {

        alert("Sem a chave da API, o site mostra o calendário incorporado.\nPara ver os cards, configure a chave.");

      }

    }
